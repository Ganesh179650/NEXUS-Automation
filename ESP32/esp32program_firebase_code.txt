// Separate FirebaseData objects to prevent socket collisions between reads and writes
FirebaseData fbdoData;     // Dedicated for outgoing telemetry JSON (ping, gas, temp, humidity, buzzer)
FirebaseData fbdoControl;  // Dedicated for reading incoming controls (servo1, servo2, led1, led2, motor)

int lastServo1Angle = -1;
int lastServo2Angle = -1;
int lastLed1 = -1;
int lastLed2 = -1;
int lastMotor = -1;

unsigned long lastControlUpdate = 0;
unsigned long lastTelemetryUpdate = 0;

const unsigned long controlInterval = 1000;   // Read controls every 1 second
const unsigned long telemetryInterval = 2000; // Update all telemetry to Firebase every 2 seconds

void connectWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  Serial.print("Connecting to WiFi");

  unsigned long start = millis();

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");

    if (millis() - start >= 30000) {
      Serial.println();
      Serial.println("WiFi connection timeout, restarting ESP32...");
      ESP.restart();
    }
  }

  Serial.println();
  Serial.println("WiFi connected");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

void setupFirebase() {
  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  config.timeout.networkReconnect = 10000;
  config.timeout.socketConnection = 30000;
  config.timeout.sslHandshake = 120000;
  config.timeout.serverResponse = 15000;

  config.timeout.rtdbKeepAlive = 45000;
  config.timeout.rtdbStreamReconnect = 10000;
  config.timeout.rtdbStreamError = 3000;

  config.tcp_data_sending_retry = 1;

  config.token_status_callback = tokenStatusCallback;

  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("Firebase authentication successful");
  } else {
    Serial.print("Firebase authentication error: ");
    Serial.println(config.signer.signupError.message.c_str());
  }

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  Serial.println("Firebase initialized");
}

void syncInitialValues() {
  int value;

  // Sync Servo 1 & move hardware motor on startup
  if (Firebase.RTDB.getInt(&fbdoControl, "/servo1")) {
    value = constrain(fbdoControl.intData(), 0, 180);
    servo1.write(value);
    lastServo1Angle = value;
    Serial.print("Initial Servo 1 synced & set to: ");
    Serial.println(value);
  } else {
    Serial.print("Initial Servo 1 sync error: ");
    Serial.println(fbdoControl.errorReason());
  }

  // Sync Servo 2 & move hardware motor on startup
  if (Firebase.RTDB.getInt(&fbdoControl, "/servo2")) {
    value = constrain(fbdoControl.intData(), 0, 180);
    servo2.write(value);
    lastServo2Angle = value;
    Serial.print("Initial Servo 2 synced & set to: ");
    Serial.println(value);
  } else {
    Serial.print("Initial Servo 2 sync error: ");
    Serial.println(fbdoControl.errorReason());
  }

  // Sync LED 1
  if (Firebase.RTDB.getInt(&fbdoControl, "/led1")) {
    lastLed1 = fbdoControl.intData();
    digitalWrite(LED1_PIN, lastLed1 ? HIGH : LOW);
    Serial.print("Initial LED 1 synced & set to: ");
    Serial.println(lastLed1 ? "ON" : "OFF");
  } else {
    Serial.print("Initial LED 1 sync error: ");
    Serial.println(fbdoControl.errorReason());
  }

  // Sync LED 2
  if (Firebase.RTDB.getInt(&fbdoControl, "/led2")) {
    lastLed2 = fbdoControl.intData();
    digitalWrite(LED2_PIN, lastLed2 ? HIGH : LOW);
    Serial.print("Initial LED 2 synced & set to: ");
    Serial.println(lastLed2 ? "ON" : "OFF");
  } else {
    Serial.print("Initial LED 2 sync error: ");
    Serial.println(fbdoControl.errorReason());
  }

  // Sync Motor
  if (Firebase.RTDB.getInt(&fbdoControl, "/motor")) {
    lastMotor = fbdoControl.intData();
    digitalWrite(MOTOR_PIN, lastMotor ? HIGH : LOW);
    Serial.print("Initial Motor synced & set to: ");
    Serial.println(lastMotor ? "ON" : "OFF");
  } else {
    Serial.print("Initial Motor sync error: ");
    Serial.println(fbdoControl.errorReason());
  }

  Serial.println("Firebase initial values synchronized successfully");
}

void checkControls() {
  int value;

  // Servo 1 Control Check
  if (Firebase.RTDB.getInt(&fbdoControl, "/servo1")) {
    value = constrain(fbdoControl.intData(), 0, 180);
    if (value != lastServo1Angle) {
      servo1.write(value);
      lastServo1Angle = value;
      Serial.print("Servo 1 moved to: ");
      Serial.println(value);
    }
  }

  // Servo 2 Control Check
  if (Firebase.RTDB.getInt(&fbdoControl, "/servo2")) {
    value = constrain(fbdoControl.intData(), 0, 180);
    if (value != lastServo2Angle) {
      servo2.write(value);
      lastServo2Angle = value;
      Serial.print("Servo 2 moved to: ");
      Serial.println(value);
    }
  }

  // LED 1 Control Check
  if (Firebase.RTDB.getInt(&fbdoControl, "/led1")) {
    value = fbdoControl.intData();
    if (value != lastLed1) {
      digitalWrite(LED1_PIN, value ? HIGH : LOW);
      lastLed1 = value;
      Serial.print("LED 1: ");
      Serial.println(value ? "ON" : "OFF");
    }
  }

  // LED 2 Control Check
  if (Firebase.RTDB.getInt(&fbdoControl, "/led2")) {
    value = fbdoControl.intData();
    if (value != lastLed2) {
      digitalWrite(LED2_PIN, value ? HIGH : LOW);
      lastLed2 = value;
      Serial.print("LED 2: ");
      Serial.println(value ? "ON" : "OFF");
    }
  }

  // Motor Fan Control Check
  if (Firebase.RTDB.getInt(&fbdoControl, "/motor")) {
    value = fbdoControl.intData();
    if (value != lastMotor) {
      digitalWrite(MOTOR_PIN, value ? HIGH : LOW);
      lastMotor = value;
      Serial.print("Motor: ");
      Serial.println(value ? "ON" : "OFF");
    }
  }
}

/**
 * Sends ALL telemetry data (ping, gas, temperature, humidity, buzzer) in ONE single 
 * atomic HTTP PATCH call using FirebaseJson.
 * This guarantees the Firebase Console UI updates live in real-time with yellow/green 
 * flashing animations without requiring a manual web browser refresh!
 */
void sendTelemetryData() {
  unsigned long uptimeSec = millis() / 1000;
  int gasValue = analogRead(GAS_PIN);
  int buzzerState = (gasValue > 1500) ? 1 : 0;
  digitalWrite(BUZZER_PIN, buzzerState ? HIGH : LOW);

  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  FirebaseJson json;
  json.set("ping", uptimeSec);
  json.set("gas", gasValue);
  json.set("buzzer", buzzerState);

  if (!isnan(temperature)) {
    json.set("temperature", temperature);
  }
  if (!isnan(humidity)) {
    json.set("humidity", humidity);
  }

  if (Firebase.RTDB.updateNode(&fbdoData, "/", &json)) {
    Serial.print("Telemetry updated live: Ping=");
    Serial.print(uptimeSec);
    Serial.print("s | Gas=");
    Serial.print(gasValue);
    if (!isnan(temperature)) {
      Serial.print(" | Temp=");
      Serial.print(temperature);
      Serial.print("C");
    }
    if (!isnan(humidity)) {
      Serial.print(" | Hum=");
      Serial.print(humidity);
      Serial.print("%");
    }
    Serial.println();
  } else {
    Serial.print("Telemetry update error: ");
    Serial.println(fbdoData.errorReason());
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(LED1_PIN, OUTPUT);
  pinMode(LED2_PIN, OUTPUT);
  pinMode(MOTOR_PIN, OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(GAS_PIN, INPUT);

  digitalWrite(LED1_PIN, LOW);
  digitalWrite(LED2_PIN, LOW);
  digitalWrite(MOTOR_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);

  servo1.setPeriodHertz(50);
  servo2.setPeriodHertz(50);

  servo1.attach(SERVO1_PIN, 500, 2400);
  servo2.attach(SERVO2_PIN, 500, 2400);

  dht.begin();

  Serial.println("Servos initialized");
  Serial.println("DHT11 initialized");
  Serial.println("Physical Buzzer on D14 initialized");

  connectWiFi();

  setupFirebase();

  delay(3000);

  syncInitialValues();

  Serial.println("System ready");
}

void loop() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi disconnected");
    connectWiFi();
  }

  if (!Firebase.ready()) {
    delay(100);
    return;
  }

  // 1. Read controls from Firebase every 1 second
  if (millis() - lastControlUpdate >= controlInterval) {
    lastControlUpdate = millis();
    checkControls();
  }

  // 2. Atomic payload push of ping, gas, temp, humidity, buzzer every 2 seconds
  if (millis() - lastTelemetryUpdate >= telemetryInterval) {
    lastTelemetryUpdate = millis();
    sendTelemetryData();
  }

  delay(10);
}
